part of 'auth_bloc.dart';

// Auth ile ilgili durumlar
abstract class AuthState extends Equatable {
  const AuthState();

  @override
  List<Object> get props => [];
}

// Başlangıç durumu
class AuthInitial extends AuthState {}

// Yükleme durumu
class AuthLoading extends AuthState {}

// Başarılı giriş/kayıt/otomatik giriş
class AuthSuccess extends AuthState {
  final String message;
  final String? authToken;
  final String? rememberMeToken;

  const AuthSuccess({required this.message, this.authToken, this.rememberMeToken});

  @override
  List<Object> get props => [message, authToken ?? '', rememberMeToken ?? ''];
}

// Hata durumu
class AuthError extends AuthState {
  final String message;

  const AuthError({required this.message});

  @override
  List<Object> get props => [message];
}

// Kullanıcı doğrulama gerektiğinde (Register'dan sonra Email doğrulamaya yönlendirmek için)
class AuthEmailVerificationRequired extends AuthState {
  final String name;
  final String email;
  final String password;
  final String passwordConfirmation;
  final String message;

  const AuthEmailVerificationRequired({
    required this.name,
    required this.email,
    required this.password,
    required this.passwordConfirmation,
    required this.message,
  });

  @override
  List<Object> get props => [name, email, password, passwordConfirmation, message];
}
