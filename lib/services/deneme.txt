  Widget build(BuildContext context) {

      theme: ThemeData(primarySwatch: Colors.blue),

class LoginPage extends StatefulWidget {
  LoginPage({super.key});

  @override
  State<LoginPage> createState() => _LoginPageState();
}

// Yeni State sınıfımız
class _LoginPageState extends State<LoginPage> {
  // ApiService instance'ımızı oluşturuyoruz
  final ApiService _apiService = ApiService();

  final TextEditingController usernameController = TextEditingController(); // Bu aslında email için kullanılacak
  final TextEditingController passwordController = TextEditingController();
  bool onPress = false;
  String username = ""; // Bu aslında email için kullanılacak
  String password = "";

  // Hata mesajlarını göstermek için yardımcı fonksiyon
  void showError(String message) {
    ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text(message)));
  }

  // Kullanıcı giriş (login) fonksiyonu
  Future<void> _loginUser() async {
    setState(() {
      onPress = true; // Butona basıldığında yükleme göstergesi gibi düşünebilirsiniz
      _message = 'Giriş yapılıyor...'; // Kullanıcıya geri bildirim
    });

    try {
      // ApiService'deki login metodunu çağırıyoruz
      // usernameController.text'in aslında email olduğunu varsayıyoruz
      final token = await _apiService.login(
        email: usernameController.text,
        password: passwordController.text,
        deviceName: 'flutter_mobile_app', // Cihaz adını burada belirtebilirsiniz
      );

      if (token != null) {
        setState(() {
          _message = 'Giriş Başarılı!';
        });
        // Giriş başarılı olursa ana sayfaya yönlendir
        Navigator.pushReplacement(context, MaterialPageRoute(builder: (context) => home_page()));
      } else {
        setState(() {
          _message = 'Giriş başarısız. Lütfen bilgilerinizi kontrol edin.';
        });
      }
    } on DioException catch (e) {
      // Dio'dan gelen spesifik hataları yakala
      String errorMessage = 'Bir hata oluştu.';
      if (e.response != null && e.response!.data != null && e.response!.data['message'] != null) {
        errorMessage = e.response!.data['message'].toString();
      } else if (e.message != null) {
        errorMessage = e.message!;
      }
      showError(errorMessage);
      setState(() {
        _message = 'Giriş Hatası: $errorMessage';
      });
    } catch (e) {
      // Diğer genel hataları yakala
      showError('Beklenmedik bir hata oluştu: ${e.toString()}');
      setState(() {
        _message = 'Beklenmedik bir hata: ${e.toString()}';
      });
    } finally {
      setState(() {
        onPress = false; // İşlem bittiğinde yükleme göstergesini kapat
      });
    }
  }

  // Kullanıcıya gösterilecek mesaj
  String _message = '';


  @override
  Widget build(BuildContext context) {
    return Scaffold(
      resizeToAvoidBottomInset: false,
      appBar: AppBar(
        title: const Text('Came Özak'),
        backgroundColor: Colors.blue,
      ),
      body: SafeArea(
        child: Center(
          child: Container(
            padding: EdgeInsets.all(20),margin:  EdgeInsets.all(9),
            decoration: BoxDecoration(color: Colors.white,borderRadius: BorderRadius.circular(15),),

            child: Column(

              mainAxisSize: MainAxisSize.min,
              children: [
                // İkon
                Container(
                  margin:  EdgeInsets.only(bottom: 20),
                  decoration: BoxDecoration(color: Colors.white70,borderRadius: BorderRadius.all(Radius.circular(20)),
                    boxShadow: [BoxShadow(color: Colors.grey.withOpacity(0.5),spreadRadius: 5, blurRadius: 15,offset: Offset(0, 3),),],),
                  child:  Icon(Icons.person,size: 150,color: Colors.blue,),
                ),
                // Kullanıcı Adı (Email olarak kullanılacak)
                Container(
                  margin:  EdgeInsets.only(bottom: 15,top: 15),
                  child: CustomTextField(
                    icon: Icons.person,
                    label: "Email Adresi", // Etiketi Email olarak değiştirdim
                    autofocus: true,
                    isPassword: false,
                    hint: "example@gmail.com",
                    controller: usernameController,
                    callback: (value){
                      print("Kullanıcı (Email): $value");
                      setState(() {
                        username = value; // username değişkeni artık email'i tutacak
                      });
                    },
                  ),
                ),
                // Şifre
                Container(
                    child: CustomTextField(
                      icon: Icons.lock,
                      label: "Şifre",
                      autofocus: false,
                      isPassword: true,
                      hint: "",
                      controller: passwordController,
                      callback: (value){
                        setState(() {
                          password = value;
                        });
                      },
                    )
                ),
                Container(
                  margin: EdgeInsets.only(bottom: 180,top: 15), alignment: Alignment.bottomRight,
                  child: GestureDetector(
                    onTap: (){Navigator.push(context, MaterialPageRoute(builder: (context) => PasswordPage(),));},
                    child: Text('Şifremi unuttum',style: TextStyle(decoration: TextDecoration.underline,color:Colors.blue)),
                  ),
                ),
                // Giriş Butonu ve Kayıt Ol Butonu
                Row(
                  children: [
                    Container(
                      width: 105,height: 40,margin: EdgeInsets.only(left: 25, bottom: 30),alignment: Alignment.bottomLeft,
                      child: CustomButton(
                        destination: home_page(), // Bu destination artık doğrudan kullanılmayacak, _loginUser içinde yönlendirme var
                        text: "Giriş yap",
                        snackText: "",
                        snack: false,
                        isNavigation: false, // Navigasyonu manuel yapacağız
                        onPress: true,
                        callback: (value){
                          if(value == "ok"){
                            _loginUser(); // API ile giriş yapma fonksiyonunu çağır
                          }
                        },
                      ),
                    ),
                    Container(
                      width: 96,height: 40,margin: EdgeInsets.only(left: 60, bottom: 30),alignment: Alignment.bottomRight,
                      child: CustomButton(
                        destination: RegisterPage(), // Kayıt sayfasına yönlendirme
                        text: 'Kayıt ol',
                        snackText: "",
                        snack: false,
                        isNavigation: true, // Navigasyon bu butonda hala geçerli
                        onPress: true,
                        callback: (value){
                          if (value == "ok"){
                            // Kayıt ol butonuna basıldığında RegisterPage'e yönlendiriyoruz.
                            // Asıl kayıt işlemi RegisterPage içinde yapılacak.
                            Navigator.push(context, MaterialPageRoute(builder: (context) => RegisterPage()));
                          }
                        },
                      ),
                    ),
                  ],
                ),
                // Mesaj gösterim alanı
                Padding(
                  padding: const EdgeInsets.all(8.0),
                  child: Text(
                    _message,
                    style: TextStyle(color: Colors.red, fontWeight: FontWeight.bold),
                    textAlign: TextAlign.center,
                  ),
                ),
              ],
            ),
          ),
        ),
      ),
    );
  }
}

import 'package:dio/dio.dart';
import 'package:shared_preferences/shared_preferences.dart';

class ApiService {
  final Dio _dio; // Dio instance'ı artık final olacak ve constructor'da initialize edilecek

  // DİKKAT: Buradaki URL'yi kendi Laravel API'nizin URL'si ile değiştirin!
  // Android emülatör kullanıyorsanız genellikle 'http://10.0.2.2:8000/api'
  // iOS simülatör veya gerçek cihaz kullanıyorsanız bilgisayarınızın yerel IP adresi
  // veya 'http://localhost:8000/api' olabilir.
  final String _baseUrl = 'http://10.0.2.2:8000/api'; // Laravel API'nizin temel URL'si

  ApiService() : _dio = Dio() { // Dio instance'ını burada initialize ediyoruz
    // Dio için temel ayarları (BaseOptions) belirliyoruz
    _dio.options = BaseOptions(
      baseUrl: _baseUrl,
      connectTimeout: const Duration(seconds: 10), // Bağlantı zaman aşımı (10 saniye)
      receiveTimeout: const Duration(seconds: 10), // Yanıt alma zaman aşımı (10 saniye)
      headers: {
        'Accept': 'application/json', // Tüm istekler için varsayılan olarak kabul başlığı
        'Content-Type': 'application/json', // Tüm istekler için varsayılan içerik tipi
      },
    );

    // Interceptor'ları Dio instance'ına ekliyoruz
    _dio.interceptors.add(InterceptorsWrapper(
      onRequest: (options, handler) async {
        final prefs = await SharedPreferences.getInstance();
        final token = prefs.getString('authToken');

        if (token != null) {
          options.headers['Authorization'] = 'Bearer $token'; // Token varsa Authorization başlığını ekle
        }
        return handler.next(options);
      },
      onError: (DioException e, handler) {
        // Hata durumunda daha detaylı loglama ve hata işleme
        print('--- Dio Hata Başlangıcı ---');
        print('İstek URL: ${e.requestOptions.uri}');
        print('Hata Tipi: ${e.type}');
        if (e.response != null) {
          print('Hata Kodu: ${e.response?.statusCode}');
          print('Hata Yanıtı: ${e.response?.data}');
        } else {
          print('Hata Mesajı: ${e.message}');
        }
        print('--- Dio Hata Sonu ---');

        // Hata tipine göre özel işlemler yapabilirsiniz
        if (e.response?.statusCode == 401) {
          // Örneğin, 401 Unauthorized hatası alırsanız tokenı temizleyebilirsiniz.
          // _clearAuthTokenAndRedirect(); // Böyle bir fonksiyon yazıp çağırabilirsiniz
          // Navigator.of(context).pushReplacement(MaterialPageRoute(builder: (_) => LoginPage())); // gibi
        } else if (e.type == DioExceptionType.connectionTimeout || e.type == DioExceptionType.receiveTimeout) {
          // Zaman aşımı hatası
          throw Exception('Ağ bağlantısı zaman aşımına uğradı. Lütfen internet bağlantınızı kontrol edin.');
        } else if (e.type == DioExceptionType.badResponse) {
          // Sunucudan hatalı yanıt (4xx, 5xx)
          throw Exception(e.response?.data['message'] ?? 'Sunucudan bir hata yanıtı alındı. Lütfen tekrar deneyin.');
        } else if (e.type == DioExceptionType.unknown) {
          // Bilinmeyen ağ hatası (internet yok, DNS sorunu vb.)
          throw Exception('Ağ bağlantısı hatası. Lütfen internet bağlantınızı kontrol edin.');
        }

        return handler.next(e); // Hatayı ilet
      },
    ));
  }

  // Kullanıcı Kayıt (Register) Metodu
  Future<Map<String, dynamic>?> register({
    String? name,
    required String email,
    required String password,
    required String passwordConfirmation,
    String deviceName = 'flutter_app',
  }) async {
    try {
      final response = await _dio.post(
        '/register', // BaseUrl zaten ayarlandığı için sadece path veriyoruz
        data: {
          if (name != null && name.isNotEmpty) 'name': name,
          'email': email,
          'password': password,
          'password_confirmation': passwordConfirmation,
          'device_name': deviceName,
        },
      );
      if (response.statusCode == 201) {
        final token = response.data['token'];
        if (token != null) {
          final prefs = await SharedPreferences.getInstance();
          await prefs.setString('authToken', token);
        }
        return response.data;
      }
    } catch (e) {
      rethrow; // Hatayı tekrar fırlatıyoruz, UI tarafında yakalanacak
    }
    return null;
  }

  // Kullanıcı Giriş (Login) Metodu
  Future<String?> login({
    required String email,
    required String password,
    String deviceName = 'flutter_app',
  }) async {
    try {
      final response = await _dio.post(
        '/login', // BaseUrl zaten ayarlandığı için sadece path veriyoruz
        data: {
          'email': email,
          'password': password,
          'device_name': deviceName,
        },
      );
      if (response.statusCode == 200) {
        final token = response.data['token'];
        if (token != null) {
          final prefs = await SharedPreferences.getInstance();
          await prefs.setString('authToken', token);
          return token;
        }
      }
    } catch (e) {
      rethrow; // Hatayı tekrar fırlatıyoruz, UI tarafında yakalanacak
    }
    return null;
  }

  // Kullanıcı Bilgilerini Alma Metodu (Korumalı Rota)
  Future<Map<String, dynamic>?> getUserData() async {
    try {
      final response = await _dio.get('/user'); // BaseUrl zaten ayarlandığı için sadece path veriyoruz
      if (response.statusCode == 200) {
        return response.data;
      }
    } catch (e) {
      rethrow; // Hatayı tekrar fırlatıyoruz, UI tarafında yakalanacak
    }
    return null;
  }

  // Kullanıcı Çıkış (Logout) Metodu
  Future<void> logout() async {
    try {
      await _dio.post('/logout'); // BaseUrl zaten ayarlandığı için sadece path veriyoruz
      final prefs = await SharedPreferences.getInstance();
      await prefs.remove('authToken');
      print('Logout successful.');
    } catch (e) {
      rethrow; // Hatayı tekrar fırlatıyoruz, UI tarafında yakalanacak
    }
  }
}
